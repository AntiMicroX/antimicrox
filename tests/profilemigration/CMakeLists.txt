cmake_minimum_required(VERSION 3.16)
project(ProfileMigrationTests LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_AUTOMOC ON)
enable_testing()
find_package(QT NAMES Qt6 Qt5 REQUIRED COMPONENTS Core Test)
find_package(Qt${QT_VERSION_MAJOR} REQUIRED COMPONENTS Core Test)
add_executable(ProfileMigrationTests testprofilemigration.cpp)
target_include_directories(ProfileMigrationTests PRIVATE ../../src)
target_link_libraries(ProfileMigrationTests PRIVATE Qt${QT_VERSION_MAJOR}::Core Qt${QT_VERSION_MAJOR}::Test)
add_test(NAME ProfileMigrationTests COMMAND ProfileMigrationTests -o -,txt -o profile-migration-results.txt,txt)
set_tests_properties(ProfileMigrationTests PROPERTIES TIMEOUT 30)
